# Live game Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A live game page showing all 10 players' rank, mastery and recent form, with a lane opponent panel.

**Architecture:** `GET /api/live/:gameName/:tagLine` returns the 10 quick cards (spectator + league + mastery, lanes guessed). The page then pulls `GET /api/live/form/:puuid?champ=` per player through a 2-worker queue, lane opponent first, and `GET /api/versus` for the lolalytics matchup.

**Tech Stack:** TypeScript, Node 22 + Express 5 (server, `tsx`), React 19 + Vite (web), vitest + supertest + Testing Library, npm workspaces.

**Spec:** `docs/superpowers/specs/2026-10-12-live-game-design.md`

## Global Constraints
- EUW only: platform `https://euw1.api.riotgames.com`, region `https://europe.api.riotgames.com`.
- Stats from Ranked Solo/Duo (queue 420) only; form uses the last 10 games.
- Lane guess only on Summoner's Rift (`mapId` 11) with 5 players per team; Smite spell id is 11.
- Anonymous players and bots: no Riot lookups, shown as "Hidden player".
- Caches: live game 30 s per Riot ID, form 10 min per PUUID + champion.
- Web form queue: 2 requests at a time; order = lane opponent, other enemies, allies.
- Badges: One-trick = 7+ of last 10 on that champion; Hot/Cold streak = 3+; First time = mastery < 10,000.
- Code style: one short comment line only where it helps; changelog one line per change.
- Run all tests: `npm test` (root). Typecheck: `npm run typecheck`.

## Review Focus
- A Riot ID whose player is in a non-SR game (ARAM): cards show, lanes null, no opponent panel, no crash.
- The searched player is anonymous in their own game (streamer mode hides their PUUID): allies are still the team holding the searched account's team id; if not found, fall back to team 100.
- Riot key expires between the game call and form calls: form cards show "Form unavailable", page keeps working.
- Clicking an enemy before its form loads: the queue picks it next, no duplicate request.
- Navigating away from the live page mid-queue: in-flight results are ignored (no state update after unmount).

```ts
import { LANES, type Lane, type MainLanesResponse } from '@lol/shared';

export const SMITE = 11;

export interface LaneCandidate { champion: { id: string }; spells: number[] }

// Smite holder is jungle; the rest get the permutation with the most main lanes
export function guessLanes(team: LaneCandidate[], mainLanes: MainLanesResponse): (Lane | null)[] {
  if (team.length !== LANES.length) return team.map(() => null);
  const lanes: (Lane | null)[] = team.map(() => null);
  const jungler = team.findIndex((p) => p.spells.includes(SMITE));
  if (jungler >= 0) lanes[jungler] = 'jungle';
  const rest = team.map((_, i) => i).filter((i) => i !== jungler);
  const free = LANES.filter((l) => jungler < 0 || l !== 'jungle');
  let best: Lane[] = free;
  let bestScore = -1;
  // first permutation is LANES order, so ties keep Riot order
  for (const perm of permutations(free)) {
    const score = perm.filter((lane, k) => mainLanes[team[rest[k]].champion.id] === lane).length;
    if (score > bestScore) { best = perm; bestScore = score; }
  }
  rest.forEach((i, k) => { lanes[i] = best[k]; });
  return lanes;
}

function permutations<T>(items: T[]): T[][] {
  if (items.length <= 1) return [items];
  return items.flatMap((x, i) => permutations([...items.slice(0, i), ...items.slice(i + 1)]).map((p) => [x, ...p]));
}
```

```ts
import type { LiveFormResponse } from '@lol/shared';
import { isRemake } from '../profile/aggregate';
import type { RiotMatch } from '../riot/types';

// matches are newest first
export function buildForm(puuid: string, matches: RiotMatch[], championKey: number): LiveFormResponse {
  const played = matches.filter((m) => !isRemake(m)).flatMap((m) => m.info.participants.filter((p) => p.puuid === puuid));
  const sum = (k: 'kills' | 'deaths' | 'assists') => played.reduce((s, p) => s + p[k], 0);
  const champ = played.filter((p) => p.championId === championKey);
  return {
    games: played.length,
    wins: played.filter((p) => p.win).length,
    streak: streak(played.map((p) => p.win)),
    kda: played.length ? Math.round(((sum('kills') + sum('assists')) / Math.max(1, sum('deaths'))) * 100) / 100 : 0,
    champGames: champ.length,
    champWins: champ.filter((p) => p.win).length,
  };
}

function streak(results: boolean[]): string {
  if (!results.length) return '';
  const end = results.findIndex((w) => w !== results[0]);
  return `${results[0] ? 'W' : 'L'}${end < 0 ? results.length : end}`;
}
```

```ts
import type { Champion, LiveFormResponse, LiveGameResponse, LivePlayer, MainLanesResponse } from '@lol/shared';
import type { TtlCache } from '../cache';
import { mapLimit } from '../mapLimit';
import type { MatchupService } from '../matchup';
import { tierForRank } from '../profile/aggregate';
import { RiotError, type RiotClient } from '../riot/client';
import type { RiotMatch, RiotSpectatorParticipant } from '../riot/types';
import { buildForm } from './form';
import { guessLanes } from './lanes';

const FORM_GAMES = 10;
const PARALLEL_FETCHES = 5;
const SUMMONERS_RIFT = 11;

export interface LiveCatalog { championByKey(key: number): Champion | undefined }

export class LiveService {
  // loads in progress, so concurrent requests share one set of Riot calls
  private pending = new Map<string, Promise<unknown>>();

  constructor(
    private riot: Pick<RiotClient, 'account' | 'activeGame' | 'leagueEntries' | 'mastery' | 'matchIds'>,
    private matches: { get(id: string): Promise<RiotMatch> },
    private catalog: LiveCatalog,
    private matchups: Pick<MatchupService, 'getMainLanes'>,
    private gameCache: TtlCache,
    private formCache: TtlCache,
  ) {}

  getGame(gameName: string, tagLine: string): Promise<LiveGameResponse> {
    return this.shared(this.gameCache, `live:${gameName.toLowerCase()}#${tagLine.toLowerCase()}`, () => this.loadGame(gameName, tagLine));
  }

  getForm(puuid: string, championKey: number): Promise<LiveFormResponse> {
    return this.shared(this.formCache, `form:${puuid}:${championKey}`, () => this.loadForm(puuid, championKey));
  }

  private shared<T>(cache: TtlCache, key: string, load: () => Promise<T>): Promise<T> {
    const running = this.pending.get(key) as Promise<T> | undefined;
    if (running) return running;
    const p = cache.getOrLoad(key, load).finally(() => this.pending.delete(key));
    this.pending.set(key, p);
    return p;
  }

  private async loadGame(gameName: string, tagLine: string): Promise<LiveGameResponse> {
    const account = await this.riot.account(gameName, tagLine);
    const game = await this.riot.activeGame(account.puuid);
    const [mainLanes, players] = await Promise.all([
      // lanes still get the Smite guess without lolalytics
      this.matchups.getMainLanes().catch((): MainLanesResponse => ({})),
      Promise.all(game.participants.map((p) => this.player(p))),
    ]);
    // hidden searched player: fall back to the blue side
    const myTeam = game.participants.find((p) => p.puuid === account.puuid)?.teamId ?? 100;
    const team = (mine: boolean) => {
      const list = players.filter((_, i) => (game.participants[i].teamId === myTeam) === mine);
      const lanes = game.mapId === SUMMONERS_RIFT ? guessLanes(list, mainLanes) : list.map(() => null);
      return list.map((p, i) => ({ ...p, lane: lanes[i] }));
    };
    const me = players.find((p) => p.puuid === account.puuid);
    return {
      queueId: game.gameQueueConfigId,
      gameStartTime: game.gameStartTime,
      me: account.puuid,
      tier: tierForRank(me?.rank?.tier),
      allies: team(true),
      enemies: team(false),
    };
  }

  private async player(p: RiotSpectatorParticipant): Promise<LivePlayer> {
    const champion = this.catalog.championByKey(p.championId) ?? { id: String(p.championId), key: p.championId, name: `#${p.championId}`, icon: '' };
    const base: LivePlayer = {
      puuid: p.puuid || null, riotId: p.riotId || null, anonymous: false, champion, spells: [p.spell1Id, p.spell2Id], lane: null,
      rank: null, seasonWins: null, seasonLosses: null, mastery: null, lookupFailed: false,
    };
    if (!p.puuid || p.bot) return { ...base, puuid: null, riotId: null, anonymous: true };
    try {
      const [entries, mastery] = await Promise.all([this.riot.leagueEntries(p.puuid), this.riot.mastery(p.puuid, p.championId)]);
      const solo = entries.find((e) => e.queueType === 'RANKED_SOLO_5x5');
      return {
        ...base, mastery,
        rank: solo ? { tier: solo.tier, division: solo.rank, lp: solo.leaguePoints } : null,
        seasonWins: solo?.wins ?? null,
        seasonLosses: solo?.losses ?? null,
      };
    } catch (e) {
      // a dead key fails the whole page; anything else only this card
      if (e instanceof RiotError && e.kind === 'key') throw e;
      return { ...base, lookupFailed: true };
    }
  }

  private async loadForm(puuid: string, championKey: number): Promise<LiveFormResponse> {
    const ids = await this.riot.matchIds(puuid, FORM_GAMES);
    const fetched = await mapLimit(ids, PARALLEL_FETCHES, (id) => this.matches.get(id).catch((e) => {
      if (e instanceof RiotError && e.kind === 'not_found') return null;
      throw e;
    }));
    return buildForm(puuid, fetched.filter((m): m is RiotMatch => m !== null), championKey);
  }
}
```

```tsx
const FIRST_TIME = 10_000;

export function laneOpponent(game: LiveGameResponse): string | null {
  const me = game.allies.find((p) => p.puuid === game.me);
  if (!me?.lane) return null;
  return game.enemies.find((p) => p.lane === me.lane && p.puuid)?.puuid ?? null;

// opponent first, then enemies, then allies
export function formOrder(game: LiveGameResponse, opponent: string | null): string[] {
  const ids = (team: LivePlayer[]) => team.flatMap((p) => (p.puuid ? [p.puuid] : []));
  const order = [...ids(game.enemies), ...ids(game.allies)];
  return opponent ? [opponent, ...order.filter((id) => id !== opponent)] : order;
}

export function badges(p: LivePlayer, form?: LiveFormResponse): string[] {
  const out: string[] = [];
  if (form && form.champGames >= ONE_TRICK) out.push('One-trick');
  const run = form ? Number(form.streak.slice(1)) : 0;
  if (form && run >= STREAK) out.push(form.streak.startsWith('W') ? 'Hot streak' : 'Cold streak');
  if (p.mastery !== null && p.mastery < FIRST_TIME) out.push('First time');
  return out;
}

  );
  return onSelect
    ? <button className="live-card" aria-pressed={selected} onClick={onSelect}>{content}</button>
    : <div className="live-card">{content}</div>;
}

export function LivePage({ riotId, onOpenChampion }: Props) {
  const [game, setGame] = useState<LiveGameResponse | null>(null);
  const [error, setError] = useState<string | null>(null);
  const [attempt, setAttempt] = useState(0);
  const [forms, setForms] = useState<Record<string, Form>>({});
  const [opponent, setOpponent] = useState<string | null>(null);
  const [versus, setVersus] = useState<VersusResponse | null | 'error' | undefined>(undefined);
  // read by the form workers, so a new opponent is fetched next
  const order = useRef<string[]>([]);
  order.current = game ? formOrder(game, opponent) : [];

  useEffect(() => {
    let cancelled = false;
    setGame(null); setError(null); setForms({}); setOpponent(null);
    const id = parseRiotId(riotId);
    const load = id ? fetchLiveGame(id.gameName, id.tagLine) : Promise.reject(new Error(`Invalid Riot ID: ${riotId}`));
    load
      .then((g) => { if (!cancelled) { setGame(g); setOpponent(laneOpponent(g)); } })
      .catch((e) => { if (!cancelled) setError((e as Error).message); });
    return () => { cancelled = true; };
  }, [riotId, attempt]);

  useEffect(() => {
    if (!game) return;
    let cancelled = false;
    const started = new Set<string>();
    const champKey = new Map([...game.allies, ...game.enemies].map((p) => [p.puuid, p.champion.key]));
    const next = () => order.current.find((id) => !started.has(id));
    const worker = async () => {
      for (let id = next(); id && !cancelled; id = next()) {
        started.add(id);
        const form: Form = await fetchLiveForm(id, champKey.get(id)!).catch(() => 'error' as const);
        if (!cancelled) setForms((f) => ({ ...f, [id]: form }));
      }
    };
    for (let i = 0; i < WORKERS; i++) void worker();
    return () => { cancelled = true; };
  }, [game]);

  const me = game?.allies.find((p) => p.puuid === game.me);
  const opp = game?.enemies.find((p) => p.puuid === opponent);
  const lane = me?.lane ?? opp?.lane ?? null;

  useEffect(() => {
    setVersus(undefined);
    if (!game || !me || !opp || !lane) return;
    let cancelled = false;
    fetchVersus(me.champion.id, opp.champion.id, lane, game.tier)
      .then((v) => { if (!cancelled) setVersus(v); })
      .catch(() => { if (!cancelled) setVersus('error'); });
    return () => { cancelled = true; };
  }, [game, me?.champion.id, opp?.champion.id, lane]);

  if (error) {
    return (
      <section className="live">
        <p className="error">{error}</p>
        <button onClick={() => setAttempt((a) => a + 1)}>Retry</button>
      </section>
    );
  }
  if (!game) return <p className="muted">Loading live game…</p>;

  const allies = byLane(game.allies);
  const enemies = byLane(game.enemies);
  const oppForm = opp?.puuid ? forms[opp.puuid] : undefined;

  return (
    <section className="live">
      <h2>Live game — {riotId}</h2>
      <div className="live-grid">
        {allies.map((a, i) => (
          <Fragment key={a.puuid ?? `a${i}`}>
            <PlayerCard player={a} form={a.puuid ? forms[a.puuid] : undefined} />
            {enemies[i] && (
              <PlayerCard
                player={enemies[i]}
                form={enemies[i].puuid ? forms[enemies[i].puuid!] : undefined}
                selected={enemies[i].puuid === opponent}
                onSelect={enemies[i].puuid ? () => setOpponent(enemies[i].puuid) : undefined}
              />
            )}
          </Fragment>
        ))}
      </div>
      {opp && me && (
        <section className="live-opponent">
          <h3>Lane opponent — {opp.riotId} ({opp.champion.name})</h3>
          <p>{formText(opp, oppForm)}</p>
          {versus === 'error' && <p className="muted">Matchup data unavailable</p>}
          {versus === null && <p className="muted">Not enough games for this matchup</p>}
          {versus && versus !== 'error' && (
            <p>{me.champion.name} vs {opp.champion.name}: {versus.winRate.toFixed(2)}% WR over {versus.games.toLocaleString('en-GB')} games</p>
          )}
          {lane && <button onClick={() => onOpenChampion(opp.champion.id, lane)}>{opp.champion.name} counters ↗</button>}
        </section>
      )}
    </section>
  );
}
```
